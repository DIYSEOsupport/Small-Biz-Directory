  <script>
    const CSV_URL = 'https://docs.google.com/spreadsheets/d/1bL_YWhS8NHDYbKBOfpz7yOIvS2Vuz4vKEbDNhXMfrsc/gviz/tq?tqx=out:csv&gid=1213716282';

    var allBusinesses = [];
    var currentCatFilter = 'ALL';

    function fetchDirectoryData(isManual) {
      var icon = document.getElementById('refreshIcon');
      if (icon) icon.classList.add('fa-spin');

      Papa.parse(CSV_URL + '&t=' + new Date().getTime(), {
        download: true,
        header: true,
        skipEmptyLines: true,
        complete: function(results) {
          if (icon) icon.classList.remove('fa-spin');
          var rows = results.data || [];

          function getRowValue(row, keywordsToMatch) {
            var keys = Object.keys(row);
            for (var i = 0; i < keys.length; i++) {
              var kRaw = keys[i].trim();
              var kLower = kRaw.toLowerCase();

              if (kLower === 'name' || kLower === 'title' || kLower.indexOf('personal page') !== -1) continue;

              for (var j = 0; j < keywordsToMatch.length; j++) {
                if (kLower.indexOf(keywordsToMatch[j].toLowerCase()) !== -1) {
                  var val = row[keys[i]];
                  if (val && val.toString().trim() !== '') {
                    return val.toString().trim();
                  }
                }
              }
            }
            return '';
          }

          var businessMap = {};

          rows.forEach(function(row) {
            var bName = getRowValue(row, ['business name']);
            if (!bName || bName.trim() === '') return;

            var key = bName.toLowerCase().trim();

            var bCat = getRowValue(row, ['category / specialty', 'category', 'specialty']);
            var bDesc = getRowValue(row, ['about your business', 'please describe your business', 'please share:']);
            var bPhone = getRowValue(row, ['business phone', 'phone']);
            var bWeb = getRowValue(row, ['business website', 'website']);
            var bFacebook = getRowValue(row, ['facebook business facebook page', 'facebook']);
            var bNextdoor = getRowValue(row, ['nextdoor business page', 'nextdoor']);
            var bEmail = getRowValue(row, ['email for customers to contact you', 'email']);
            var bLoc = getRowValue(row, ['business city', 'area / city served', 'business street address', 'business address', 'business state', 'business zip']);
            var bSlogan = getRowValue(row, ['additional info or slogan', 'slogan']);

            if (!businessMap[key]) {
              businessMap[key] = {
                name: bName.trim(),
                category: bCat,
                description: bDesc,
                phone: bPhone,
                website: bWeb,
                facebook: bFacebook,
                nextdoor: bNextdoor,
                email: bEmail,
                location: bLoc,
                slogan: bSlogan
              };
            } else {
              if (!businessMap[key].category && bCat) businessMap[key].category = bCat;
              if (!businessMap[key].description && bDesc) businessMap[key].description = bDesc;
              if (!businessMap[key].phone && bPhone) businessMap[key].phone = bPhone;
              if (!businessMap[key].website && bWeb) businessMap[key].website = bWeb;
              if (!businessMap[key].facebook && bFacebook) businessMap[key].facebook = bFacebook;
              if (!businessMap[key].nextdoor && bNextdoor) businessMap[key].nextdoor = bNextdoor;
              if (!businessMap[key].email && bEmail) businessMap[key].email = bEmail;
              if (!businessMap[key].location && bLoc) businessMap[key].location = bLoc;
              if (!businessMap[key].slogan && bSlogan) businessMap[key].slogan = bSlogan;
            }
          });

          allBusinesses = Object.values(businessMap).map(function(b) {
            return {
              name: b.name,
              category: b.category || 'Local Business',
              description: b.description || 'No detailed description provided yet.',
              phone: b.phone || '',
              website: b.website || '',
              facebook: b.facebook || '',
              nextdoor: b.nextdoor || '',
              email: b.email || '',
              location: b.location || 'Charlotte County, FL',
              slogan: b.slogan || ''
            };
          });

          document.getElementById('loading-spinner').style.display = 'none';
          filterDirectory();

          var now = new Date().toLocaleTimeString([], { hour: '2-digit', minute: '2-digit', second: '2-digit' });
          document.getElementById('syncStatus').innerHTML = '<i class="fa-solid fa-circle-check" style="color:#16a34a;"></i> Synchronized at ' + now + ' (Auto-refreshes every 60s)';
        },
        error: function(err) {
          if (icon) icon.classList.remove('fa-spin');
          document.getElementById('loading-spinner').innerHTML = '<div style="color:#ef4444; padding:20px;">Unable to load spreadsheet records. Verify Google Sheet sharing is set to "Anyone with the link".</div>';
        }
      });
    }

    fetchDirectoryData(false);

    setInterval(function() {
      fetchDirectoryData(false);
    }, 60000);

    function renderDirectory(list) {
      var grid = document.getElementById('directoryGrid');
      grid.innerHTML = '';
      if (!list || list.length === 0) {
        grid.innerHTML = '<div class="empty-state"><i class="fa-solid fa-store-slash" style="font-size:32px; margin-bottom:12px; color:var(--primary-light);"></i><h3>No matching business listings found</h3><p>Try adjusting your search query or category filter above.</p></div>';
        return;
      }
      list.forEach(function(b) {
        var name = escapeHtml(b.name);
        var category = escapeHtml(b.category);
        var location = escapeHtml(b.location);
        var slogan = escapeHtml(b.slogan);
        var description = escapeHtml(b.description);
        var phone = escapeHtml(b.phone);
        var website = b.website;
        var facebook = b.facebook;
        var nextdoor = b.nextdoor;
        var email = escapeHtml(b.email);
        var initial = name.charAt(0).toUpperCase();

        var html = '<div class="biz-card"><div>';
        html += '<div class="card-header"><div class="avatar-icon">' + initial + '</div><div class="biz-title">' + name + '</div></div>';

        html += '<div class="tag-container">';
        html += '<span class="tag tag-cat"><i class="fa-solid fa-tag"></i> ' + category + '</span>';
        html += '<span class="tag tag-loc"><i class="fa-solid fa-location-dot"></i> ' + location + '</span>';
        html += '</div>';

        if (slogan) {
          html += '<div class="biz-slogan">"' + slogan + '"</div>';
        } else {
          html += '<div class="biz-slogan text-placeholder">"Serving Charlotte County Community"</div>';
        }

        html += '<div class="biz-desc">' + description + '</div>';
        html += '</div>';

        html += '<div class="card-actions">';
        
        html += '<div class="action-row">';
        if (phone) {
          html += '<a href="tel:' + phone + '" class="btn-action btn-phone"><i class="fa-solid fa-phone"></i> Direct Line</a>';
        } else {
          html += '<span class="btn-action btn-disabled"><i class="fa-solid fa-phone-slash"></i> Phone Unlisted</span>';
        }

        if (website) {
          html += '<a href="' + formatUrl(website) + '" target="_blank" class="btn-action btn-web"><i class="fa-solid fa-globe"></i> Official Web Portal</a>';
        } else {
          html += '<span class="btn-action btn-disabled"><i class="fa-solid fa-globe"></i> Website Unlisted</span>';
        }
        html += '</div>';

        if (facebook || nextdoor || email) {
          html += '<div class="action-row">';
          if (facebook) {
            html += '<a href="' + formatUrl(facebook) + '" target="_blank" class="btn-action btn-social btn-facebook"><i class="fa-brands fa-facebook-f"></i> Facebook Page</a>';
          }
          if (nextdoor) {
            html += '<a href="' + formatUrl(nextdoor) + '" target="_blank" class="btn-action btn-social btn-nextdoor"><i class="fa-solid fa-house-user"></i> Nextdoor Profile</a>';
          }
          if (email) {
            html += '<a href="mailto:' + email + '" class="btn-action btn-social btn-email"><i class="fa-solid fa-envelope"></i> Email Contact</a>';
          }
          html += '</div>';
        }

        html += '</div>';
        html += '</div>';
        grid.innerHTML += html;
      });
    }

    function setCategoryFilter(category, btnElement) {
      currentCatFilter = category;
      document.querySelectorAll('.pill-btn').forEach(b => b.classList.remove('active'));
      btnElement.classList.add('active');
      filterDirectory();
    }

    function filterDirectory() {
      var query = document.getElementById('searchInput').value.toLowerCase();
      var filtered = allBusinesses.filter(function(b) {
        var matchesQuery = (b.name || '').toLowerCase().indexOf(query) !== -1 ||
                            (b.category || '').toLowerCase().indexOf(query) !== -1 ||
                            (b.description || '').toLowerCase().indexOf(query) !== -1 ||
                            (b.location || '').toLowerCase().indexOf(query) !== -1;
        var matchesCategory = (currentCatFilter === 'ALL') || 
                              ((b.category || '') + ' ' + (b.description || '')).toLowerCase().indexOf(currentCatFilter.toLowerCase()) !== -1;
        return matchesQuery && matchesCategory;
      });
      renderDirectory(filtered);
    }

    function escapeHtml(str) { return (str || '').toString().replace(/&/g, "&amp;").replace(/</g, "&lt;").replace(/>/g, "&gt;"); }
    function formatUrl(url) { if (!url) return '#'; url = url.toString().trim(); return (url.indexOf('http') === 0) ? url : 'https://' + url; }
  </script>
