  <script>
    // Targeted directly to the "Public Directory" tab by name
    const CSV_URL = 'https://docs.google.com/spreadsheets/d/1bL_YWhS8NHDYbKBOfpz7yOIvS2Vuz4vKEbDNhXMfrsc/gviz/tq?tqx=out:csv&sheet=Public%20Directory';

    var allBusinesses = [];
    var currentCatFilter = 'ALL';

    function isEmail(str) {
      return str && str.indexOf('@') !== -1 && str.indexOf('.') !== -1;
    }

    function isUrl(str) {
      if (!str) return false;
      var s = str.toLowerCase().trim();
      return (s.indexOf('http') === 0 || s.indexOf('www.') === 0 || s.indexOf('.com') !== -1 || s.indexOf('.org') !== -1 || s.indexOf('.net') !== -1) && !isEmail(s);
    }

    function isPhone(str) {
      if (!str) return false;
      var digits = str.replace(/\D/g, '');
      return digits.length >= 7 && !isEmail(str);
    }

    function fetchDirectoryData(isManual) {
      var icon = document.getElementById('refreshIcon');
      if (icon) icon.classList.add('fa-spin');

      Papa.parse(CSV_URL + '&t=' + new Date().getTime(), {
        download: true,
        header: true,
        skipEmptyLines: true,
        complete: function(results) {
          if (icon) icon.classList.remove('fa-spin');
          var rows = results.data || [];

          function findRawValue(row, keywords) {
            var keys = Object.keys(row);
            for (var i = 0; i < keys.length; i++) {
              var kLower = keys[i].trim().toLowerCase();
              if (kLower === 'timestamp' || kLower === 'name' || kLower === 'title' || kLower.indexOf('personal page') !== -1) continue;

              for (var j = 0; j < keywords.length; j++) {
                if (kLower.indexOf(keywords[j].toLowerCase()) !== -1) {
                  var val = row[keys[i]];
                  if (val && val.toString().trim() !== '' && val.toString().trim() !== '-') {
                    return val.toString().trim();
                  }
                }
              }
            }
            return '';
          }

          var businessMap = {};

          rows.forEach(function(row) {
            var bName = findRawValue(row, ['business name']);
            
            if (!bName || bName.length < 2 || bName === '-' || bName.toLowerCase() === 'test' || bName.toLowerCase() === 'none' || bName.toLowerCase().indexOf('business name') !== -1) return;

            var key = bName.toLowerCase().trim();

            var bCat = findRawValue(row, ['category / specialty', 'category', 'specialty']);
            var bDescRaw = findRawValue(row, ['please describe your business', 'about your business', 'please share:']);
            var bPhoneRaw = findRawValue(row, ['business phone', 'phone', 'telephone']);
            var bWebRaw = findRawValue(row, ['business website', 'website', 'domain']);
            var bFacebookRaw = findRawValue(row, ['facebook business facebook page', 'facebook']);
            var bNextdoorRaw = findRawValue(row, ['nextdoor business page', 'nextdoor']);
            var bEmailRaw = findRawValue(row, ['email for customers', 'customer email']);
            var bLocRaw = findRawValue(row, ['area / city served', 'business city', 'city']);
            var bSloganRaw = findRawValue(row, ['additional info or slogan', 'slogan', 'tagline']);

            var phone = isPhone(bPhoneRaw) ? bPhoneRaw : '';
            var email = isEmail(bEmailRaw) ? bEmailRaw : (isEmail(bPhoneRaw) ? bPhoneRaw : (isEmail(bWebRaw) ? bWebRaw : (isEmail(bSloganRaw) ? bSloganRaw : '')));
            var website = isUrl(bWebRaw) ? bWebRaw : (isUrl(bSloganRaw) ? bSloganRaw : '');
            var slogan = (!isEmail(bSloganRaw) && !isUrl(bSloganRaw) && !isPhone(bSloganRaw)) ? bSloganRaw : '';
            var facebook = (bFacebookRaw && bFacebookRaw.toLowerCase().indexOf('facebook') !== -1) ? bFacebookRaw : (isUrl(bFacebookRaw) ? bFacebookRaw : '');
            var nextdoor = (bNextdoorRaw && bNextdoorRaw.toLowerCase().indexOf('nextdoor') !== -1) ? bNextdoorRaw : (isUrl(bNextdoorRaw) ? bNextdoorRaw : '');

            var location = bLocRaw;
            if (!location || isPhone(location) || isEmail(location)) {
              location = 'Charlotte County, FL';
            }

            var operatingMode = '';
            if (bDescRaw === 'Physical Location' || bDescRaw === 'Online Only' || bDescRaw === 'Mobile Service' || bDescRaw.indexOf('Mobile Service') !== -1) {
              operatingMode = bDescRaw;
              bDescRaw = '';
            }

            if (!businessMap[key]) {
              businessMap[key] = {
                name: bName.trim(),
                category: bCat || 'Local Business',
                operatingMode: operatingMode,
                description: bDescRaw,
                phone: phone,
                website: website,
                facebook: facebook,
                nextdoor: nextdoor,
                email: email,
                location: location,
                slogan: slogan
              };
            } else {
              if (!businessMap[key].category && bCat) businessMap[key].category = bCat;
              if (!businessMap[key].operatingMode && operatingMode) businessMap[key].operatingMode = operatingMode;
              if (!businessMap[key].description && bDescRaw) businessMap[key].description = bDescRaw;
              if (!businessMap[key].phone && phone) businessMap[key].phone = phone;
              if (!businessMap[key].website && website) businessMap[key].website = website;
              if (!businessMap[key].facebook && facebook) businessMap[key].facebook = facebook;
              if (!businessMap[key].nextdoor && nextdoor) businessMap[key].nextdoor = nextdoor;
              if (!businessMap[key].email && email) businessMap[key].email = email;
              if (!businessMap[key].location && location) businessMap[key].location = location;
              if (!businessMap[key].slogan && slogan) businessMap[key].slogan = slogan;
            }
          });

          allBusinesses = Object.values(businessMap);

          document.getElementById('loading-spinner').style.display = 'none';
          updateCategoryPills();
          filterAndSortDirectory();

          var now = new Date().toLocaleTimeString([], { hour: '2-digit', minute: '2-digit', second: '2-digit' });
          document.getElementById('syncStatus').innerHTML = '<i class="fa-solid fa-circle-check" style="color:#16a34a;"></i> Synchronized at ' + now + ' (Auto-refreshes every 60s)';
        },
        error: function(err) {
          if (icon) icon.classList.remove('fa-spin');
          document.getElementById('loading-spinner').innerHTML = '<div style="color:#ef4444; padding:20px;">Unable to load spreadsheet records. Verify Google Sheet sharing is set to "Anyone with the link".</div>';
        }
      });
    }

    fetchDirectoryData(false);

    setInterval(function() {
      fetchDirectoryData(false);
    }, 60000);

    function updateCategoryPills() {
      var container = document.getElementById('filterPills');
      var categories = {};
      
      allBusinesses.forEach(function(b) {
        if (b.category && b.category.trim() !== '') {
          categories[b.category.trim()] = true;
        }
      });

      var catList = Object.keys(categories).sort();
      
      var html = '<button class="pill-btn ' + (currentCatFilter === 'ALL' ? 'active' : '') + '" onclick="setCategoryFilter(\'ALL\', this)"><i class="fa-solid fa-border-all"></i> All Listings (' + allBusinesses.length + ')</button>';

      catList.forEach(function(cat) {
        var count = allBusinesses.filter(function(b) { return b.category === cat; }).length;
        var isActive = currentCatFilter === cat ? 'active' : '';
        html += '<button class="pill-btn ' + isActive + '" onclick="setCategoryFilter(\'' + escapeJs(cat) + '\', this)"><i class="fa-solid fa-tag"></i> ' + escapeHtml(cat) + ' (' + count + ')</button>';
      });

      container.innerHTML = html;
    }

    function setCategoryFilter(category, btnElement) {
      currentCatFilter = category;
      document.querySelectorAll('.pill-btn').forEach(b => b.classList.remove('active'));
      if (btnElement) btnElement.classList.add('active');
      filterAndSortDirectory();
    }

    function filterAndSortDirectory() {
      var query = document.getElementById('searchInput').value.toLowerCase().trim();
      var sortBy = document.getElementById('sortSelect').value;

      var filtered = allBusinesses.filter(function(b) {
        var matchesQuery = !query || 
                            (b.name || '').toLowerCase().indexOf(query) !== -1 ||
                            (b.category || '').toLowerCase().indexOf(query) !== -1 ||
                            (b.description || '').toLowerCase().indexOf(query) !== -1 ||
                            (b.location || '').toLowerCase().indexOf(query) !== -1 ||
                            (b.slogan || '').toLowerCase().indexOf(query) !== -1 ||
                            (b.operatingMode || '').toLowerCase().indexOf(query) !== -1;

        var matchesCategory = (currentCatFilter === 'ALL') || (b.category === currentCatFilter);

        return matchesQuery && matchesCategory;
      });

      var sorted = sortList(filtered, sortBy);
      renderDirectory(sorted);
    }

    function sortList(list, sortBy) {
      var sorted = list.slice();
      if (sortBy === 'name-asc') {
        sorted.sort((a, b) => a.name.localeCompare(b.name));
      } else if (sortBy === 'name-desc') {
        sorted.sort((a, b) => b.name.localeCompare(a.name));
      } else if (sortBy === 'cat-asc') {
        sorted.sort((a, b) => a.category.localeCompare(b.category));
      } else if (sortBy === 'loc-asc') {
        sorted.sort((a, b) => a.location.localeCompare(b.location));
      }
      return sorted;
    }

    function renderDirectory(list) {
      var grid = document.getElementById('directoryGrid');
      grid.innerHTML = '';
      if (!list || list.length === 0) {
        grid.innerHTML = '<div class="empty-state"><i class="fa-solid fa-store-slash" style="font-size:32px; margin-bottom:12px; color:var(--primary-light);"></i><h3>No matching business listings found</h3><p>Try adjusting your search query or category filter above.</p></div>';
        return;
      }
      list.forEach(function(b) {
        var name = escapeHtml(b.name);
        var category = escapeHtml(b.category);
        var operatingMode = escapeHtml(b.operatingMode);
        var location = escapeHtml(b.location);
        var slogan = escapeHtml(b.slogan);
        var description = escapeHtml(b.description);
        var phone = escapeHtml(b.phone);
        var website = b.website;
        var facebook = b.facebook;
        var nextdoor = b.nextdoor;
        var email = escapeHtml(b.email);
        var initial = name.charAt(0).toUpperCase();

        var html = '<div class="biz-card"><div>';
        html += '<div class="card-header"><div class="avatar-icon">' + initial + '</div><div class="biz-title">' + name + '</div></div>';

        html += '<div class="tag-container">';
        if (category) html += '<span class="tag tag-cat"><i class="fa-solid fa-tag"></i> ' + category + '</span>';
        if (location) html += '<span class="tag tag-loc"><i class="fa-solid fa-location-dot"></i> ' + location + '</span>';
        if (operatingMode) html += '<span class="tag tag-mode"><i class="fa-solid fa-building"></i> ' + operatingMode + '</span>';
        html += '</div>';

        if (slogan) {
          html += '<div class="biz-slogan">"' + slogan + '"</div>';
        }

        if (description) {
          html += '<div class="biz-desc">' + description + '</div>';
        }

        html += '</div>';

        if (phone || website || facebook || nextdoor || email) {
          html += '<div class="card-actions">';
          html += '<div class="action-row">';
          
          if (phone) {
            html += '<a href="tel:' + phone + '" class="btn-action btn-phone"><i class="fa-solid fa-phone"></i> Direct Line</a>';
          }
          if (website) {
            html += '<a href="' + formatUrl(website) + '" target="_blank" class="btn-action btn-web"><i class="fa-solid fa-globe"></i> Official Web Portal</a>';
          }
          if (facebook) {
